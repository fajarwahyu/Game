<script>
    // Database Pertanyaan Jaringan SMK Kelas 10
    const questions = [
        {
            q: "Perangkat keras yang berfungsi untuk menghubungkan beberapa komputer dalam satu jaringan lokal (LAN) adalah...",
            options: ["Modem", "Switch", "Router", "Repeater"],
            answer: 1 // Index jawaban benar (Switch)
        },
        {
            q: "Alamat unik yang diberikan kepada setiap perangkat dalam jaringan komputer disebut...",
            options: ["IP Address", "MAC Address", "URL", "DNS"],
            answer: 0 // IP Address
        },
        {
            q: "Kabel jaringan yang paling umum digunakan saat ini dengan konektor RJ-45 adalah kabel...",
            options: ["Coaxial", "Fiber Optic", "UTP (Twisted Pair)", "USB"],
            answer: 2 // UTP
        },
        {
            q: "Topologi jaringan di mana semua komputer terhubung ke satu kabel pusat disebut topologi...",
            options: ["Star", "Ring", "Bus", "Mesh"],
            answer: 2 // Bus
        },
        {
            q: "Protokol standar yang digunakan untuk browsing website di internet adalah...",
            options: ["FTP", "SMTP", "HTTP/HTTPS", "SSH"],
            answer: 2 // HTTP
        }
    ];


    let currentQuestionIndex = 0;
    let ropePosition = 50; // 0 = Menang Siswa, 100 = Menang Komputer, 50 = Tengah
    let isGameOver = false;


    const questionText = document.getElementById('questionText');
    const optionsContainer = document.getElementById('optionsContainer');
    const feedbackText = document.getElementById('feedbackText');
    const ropeWrapper = document.getElementById('ropeWrapper');
    const restartBtn = document.getElementById('restartBtn');
    const scoreStudentEl = document.getElementById('score-student');
    const scoreCompEl = document.getElementById('score-comp');


    function initGame() {
        currentQuestionIndex = 0;
        ropePosition = 50;
        isGameOver = false;
        updateRopeVisual();
        restartBtn.style.display = 'none';
        scoreStudentEl.innerText = '0';
        scoreCompEl.innerText = '0';
        feedbackText.innerText = '';
        loadQuestion();
    }


    function updateRopeVisual() {
        // Menggerakkan wrapper. 
        // Jika ropePosition 50 -> left: 50%
        // Jika ropePosition 0 (Siswa menang) -> left: 0% (geser ke kiri)
        // Jika ropePosition 100 (Komputer menang) -> left: 100% (geser ke kanan)
        // Namun, karena wrapper lebar 100%, kita perlu logika sedikit berbeda agar simpul (knot) berada di posisi yang pas.
        
        // Kita gunakan logika sederhana: Geser 'left' dari wrapper.
        // Tapi agar knot terlihat bergerak mulus dari tengah ke kiri/kanan container:
        // Container lebar 100%. Knot di tengah wrapper.
        // Agar knot ada di 0% container, wrapper left harus -50%? Tidak, mari sederhanakan.
        
        // Cara termudah: Ubah left dari ropeWrapper.
        // Default left: 50%. 
        // Step per soal: 100% / 5 soal = 20% per geseran.
        // Benar: kurangi left (ke kiri). Salah: tambah left (ke kanan).
        
        ropeWrapper.style.left = ropePosition + "%";
    }


    function loadQuestion() {
        if (currentQuestionIndex >= questions.length) {
            endGame(true); // Semua soal habis
            return;
        }


        const currentQ = questions[currentQuestionIndex];
        questionText.innerText = `Soal ${currentQuestionIndex + 1}: ${currentQ.q}`;
        optionsContainer.innerHTML = '';


        currentQ.options.forEach((opt, index) => {
            const btn = document.createElement('button');
            btn.className = 'option-btn';
            btn.innerText = opt;
            btn.onclick = () => checkAnswer(index);
            optionsContainer.appendChild(btn);
        });
    }


    function checkAnswer(selectedIndex) {
        if (isGameOver) return;


        const currentQ = questions[currentQuestionIndex];
        const buttons = document.querySelectorAll('.option-btn');
        
        // Disable semua tombol
        buttons.forEach(btn => btn.disabled = true);


        if (selectedIndex === currentQ.answer) {
            // Jawaban BENAR
            feedbackText.innerText = "Benar! Tarik ke kiri!";
            feedbackText.className = "feedback correct";
            buttons[selectedIndex].style.backgroundColor = "#27ae60";
            
            // Logika Tarik Tambang: Geser ke KIRI (kurangi persentase)
            ropePosition -= 10; // Geser 10% ke kiri
            scoreStudentEl.innerText = parseInt(scoreStudentEl.innerText) + 1;


        } else {
            // Jawaban SALAH
            feedbackText.innerText = "Salah! Komputer menarik ke kanan!";
            feedbackText.className = "feedback wrong";
            buttons[selectedIndex].style.backgroundColor = "#c0392b";
            buttons[currentQ.answer].style.backgroundColor = "#27ae60"; // Tunjukkan yang benar


            // Logika Tarik Tambang: Geser ke KANAN (tambah persentase)
            ropePosition += 10; // Geser 10% ke kanan
            scoreCompEl.innerText = parseInt(scoreCompEl.innerText) + 1;
        }


        updateRopeVisual();
        checkGameState();
    }


    function checkGameState() {
        // Cek kondisi menang/kalah berdasarkan posisi tali
        // Jika ropePosition <= 10 (Sangat ke kiri), Siswa Menang Mutlak
        // Jika ropePosition >= 90 (Sangat ke kanan), Komputer Menang Mutlak
        
        setTimeout(() => {
            if (ropePosition <= 10) {
                endGame(true, "Selamat! Kamu menguasai dasar jaringan!");
            } else if (ropePosition >= 90) {
                endGame(false, "Yah, Komputer menang. Ayo belajar lagi!");
            } else {
                // Lanjut ke soal berikutnya jika belum habis
                currentQuestionIndex++;
                feedbackText.innerText = ""; // Reset feedback text agar tidak menumpuk
                loadQuestion();
            }
        }, 1000); // Delay 1 detik agar siswa melihat hasil
    }


    function endGame(isWin, message) {
        isGameOver = true;
        optionsContainer.innerHTML = '';
        questionText.innerText = isWin ? "🏆 VICTORY 🏆" : "💀 DEFEAT 💀";
        feedbackText.innerText = message || (isWin ? "Luar biasa! Semua jawaban benar." : "Jangan menyerah, coba lagi!");
        feedbackText.className = isWin ? "feedback correct" : "feedback wrong";
        restartBtn.style.display = 'inline-block';
    }


    // Mulai game saat load
    initGame();


</script>
